# Windows version resource for Plunder.exe (PyInstaller --version-file).
#
# Why this file exists: an unsigned PyInstaller executable with no version
# resource at all is one of the shapes Microsoft Defender's generic
# heuristics score as suspicious (it is what freshly packed malware looks
# like). Real metadata does not make the exe trusted, but it removes one of
# the cheap signals that push an otherwise ordinary program over the line
# into a "Trojan:Win32/Wacatac"-style detection.
#
# Bump the four numbers together when the version changes - filevers and
# prodvers are tuples, FileVersion and ProductVersion are the strings that
# actually show up in the file's Properties dialog.

    filevers=(1, 0, 0, 0),
    prodvers=(1, 0, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          '040904B0',

          ]
        )
      ]
    ),

  ]
